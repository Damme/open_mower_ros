#include "ros/ros.h"
#include "std_msgs/String.h"
#include "wiringPiI2C.h"
#include <sstream>
#include <cmath>

#define MPU6050_ADDR 0x68

#define ACCEL_CONFIG 0x1C
#define GYRO_CONFIG 0x1B
#define ACCEL_XOUT_H 0x3B
#define ACCEL_YOUT_H 0x3D
#define ACCEL_ZOUT_H 0x3F
#define GYRO_XOUT_H 0x43
#define GYRO_YOUT_H 0x45
#define GYRO_ZOUT_H 0x47

#define ACCEL_SCALE_MODIFIER_2G 16384.0
#define GYRO_SCALE_MODIFIER_200DEG 65.5

#define GRAVITY 9.81686

int16_t read_word_2c(int fd, int addr) {
    int16_t high = wiringPiI2CReadReg8(fd, addr);
    int16_t low = wiringPiI2CReadReg8(fd, addr + 1);
    int16_t value = (high << 8) + low;
    if (value >= 0x8000) {
        value = -(65536 - value);
    }
    return value;
}

void calibrate_sensor(int fd, int samples, double &accel_x_offset, double &accel_y_offset, double &accel_z_offset, double &gyro_x_offset, double &gyro_y_offset, double &gyro_z_offset) {
    double accel_x_sum = 0, accel_y_sum = 0, accel_z_sum = 0;
    double gyro_x_sum = 0, gyro_y_sum = 0, gyro_z_sum = 0;

    for (int i = 0; i < samples; i++) {
        accel_x_sum += read_word_2c(fd, ACCEL_XOUT_H);
        accel_y_sum += read_word_2c(fd, ACCEL_YOUT_H);
        accel_z_sum += read_word_2c(fd, ACCEL_ZOUT_H);
        gyro_x_sum += read_word_2c(fd, GYRO_XOUT_H);
        gyro_y_sum += read_word_2c(fd, GYRO_YOUT_H);
        gyro_z_sum += read_word_2c(fd, GYRO_ZOUT_H);

        ros::Duration(0.01).sleep();
    }

    accel_x_offset = accel_x_sum / samples;
    accel_y_offset = accel_y_sum / samples;
    accel_z_offset = accel_z_sum / samples - ACCEL_SCALE_MODIFIER_2G; // Account for gravity
    gyro_x_offset = gyro_x_sum / samples;
    gyro_y_offset = gyro_y_sum / samples;
    gyro_z_offset = gyro_z_sum / samples;
}

int main(int argc, char **argv) {
    ros::init(argc, argv, "mpu6050_node");
    ros::NodeHandle nh;
    ros::Publisher pub = nh.advertise<std_msgs::String>("mpu6050_data", 1000);
    ros::Rate loop_rate(10);

    int fd = wiringPiI2CSetup(MPU6050_ADDR);
    if (fd == -1) {
        ROS_ERROR("Failed to initialize I2C communication.");
        return -1;
    }
    ROS_INFO("I2C communication successfully initialized.");

    wiringPiI2CWriteReg8(fd, 0x6B, 0x00); // Wake up MPU6050
    wiringPiI2CWriteReg8(fd, ACCEL_CONFIG, 0x00); // ±2g
    wiringPiI2CWriteReg8(fd, GYRO_CONFIG, 0x08);  // 200°/s

    // Calibration
    double accel_x_offset, accel_y_offset, accel_z_offset;
    double gyro_x_offset, gyro_y_offset, gyro_z_offset;
    ROS_INFO("Calibrating... Please keep the sensor stationary.");
    calibrate_sensor(fd, 100, accel_x_offset, accel_y_offset, accel_z_offset, gyro_x_offset, gyro_y_offset, gyro_z_offset);
    ROS_INFO("Calibration complete.");

    while (ros::ok()) {
        int16_t accel_x_raw = read_word_2c(fd, ACCEL_XOUT_H);
        int16_t accel_y_raw = read_word_2c(fd, ACCEL_YOUT_H);
        int16_t accel_z_raw = read_word_2c(fd, ACCEL_ZOUT_H);
        int16_t gyro_x_raw = read_word_2c(fd, GYRO_XOUT_H);
        int16_t gyro_y_raw = read_word_2c(fd, GYRO_YOUT_H);
        int16_t gyro_z_raw = read_word_2c(fd, GYRO_ZOUT_H);

        // Convert raw values to acceleration in m/s² and angular velocity in rad/s
        double accel_x = ((accel_x_raw - accel_x_offset) / ACCEL_SCALE_MODIFIER_2G) * GRAVITY;
        double accel_y = ((accel_y_raw - accel_y_offset) / ACCEL_SCALE_MODIFIER_2G) * GRAVITY;
        double accel_z = ((accel_z_raw - accel_z_offset) / ACCEL_SCALE_MODIFIER_2G) * GRAVITY;
        double gyro_x = ((gyro_x_raw - gyro_x_offset) / GYRO_SCALE_MODIFIER_200DEG) * (M_PI / 180.0);
        double gyro_y = ((gyro_y_raw - gyro_y_offset) / GYRO_SCALE_MODIFIER_200DEG) * (M_PI / 180.0);
        double gyro_z = ((gyro_z_raw - gyro_z_offset) / GYRO_SCALE_MODIFIER_200DEG) * (M_PI / 180.0);

        std_msgs::String msg;
        std::stringstream ss;
        ss << "Accel: [" << accel_x << ", " << accel_y << ", " << accel_z << "], "
           << "Gyro: [" << gyro_x << ", " << gyro_y << ", " << gyro_z << "]";
        msg.data = ss.str();

        pub.publish(msg);
        ros::spinOnce();
        loop_rate.sleep();
    }

    return 0;
}
