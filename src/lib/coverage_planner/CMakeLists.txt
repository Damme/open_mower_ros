cmake_minimum_required(VERSION 3.0.2)
project(coverage_planner)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# -DNDEBUG disables Boost.Geometry asserts; the rest shrink the binary.
add_compile_options(-O2 -DNDEBUG
  -ffunction-sections -fdata-sections
  -fvisibility=hidden -fvisibility-inlines-hidden)

find_package(catkin REQUIRED COMPONENTS
        geometry_msgs
        nav_msgs
        roscpp
        std_msgs
        message_generation
        tf2
        tf2_geometry_msgs
        visualization_msgs
        )

# Boost.Geometry is header-only, so no linked components are required.
find_package(Boost REQUIRED)

## Messages / services generated by THIS package.
add_message_files(
        FILES
        Path.msg
)

add_service_files(
        FILES
        PlanPath.srv
)

generate_messages(
        DEPENDENCIES
        geometry_msgs
        nav_msgs
        std_msgs
)

catkin_package(
        CATKIN_DEPENDS
        message_runtime
        geometry_msgs
        nav_msgs
        std_msgs
        roscpp
        tf2
        tf2_geometry_msgs
        visualization_msgs
)

include_directories(
        ${catkin_INCLUDE_DIRS}
        ${Boost_INCLUDE_DIRS}
)

add_executable(coverage_planner
        src/coverage_planner.cpp
        src/geom.hpp
        )

# Build this package's generated message/service headers before the node.
add_dependencies(coverage_planner
        ${catkin_EXPORTED_TARGETS}
        ${${PROJECT_NAME}_EXPORTED_TARGETS}
        )

# THIS is the line that was lost with the slic3r block -> all the ros::*
# undefined references. roscpp/rosconsole/serialization come in via catkin.
target_link_libraries(coverage_planner
        ${catkin_LIBRARIES}
        ${Boost_LIBRARIES}
        )

install(TARGETS coverage_planner
        RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
)

catkin_install_python(PROGRAMS
  scripts/plan_and_dump.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
)
